### Location: `shared/lib/apollo-client/`

#### Client-Side Setup (ApolloClientProvider):

```typescript
// Use in: app/provider/RootProviders.tsx
import { ApolloClientProvider } from "@/shared/lib";

function makeClient() {
  const httpLink = new HttpLink({
    uri: process.env.NEXT_PUBLIC_BASE_URL + "/api/graphql",
    fetchOptions: {
      cache: "no-store",              // Always fresh data
      credentials: "same-origin",     // Include cookies
    },
    credentials: "same-origin",
  });

  return new ApolloClient({
    cache: new InMemoryCache(),
    link: httpLink,
  });
}

export function ApolloClientProvider({ children }) {
  return (
    <ApolloNextAppProvider makeClient={makeClient}>
      {children}
    </ApolloNextAppProvider>
  );
}
```

**Key Configuration**:

- Cache: `InMemoryCache` (no persistence)
- Fetch: `no-store` (disable HTTP caching)
- Auth: Cookies via `credentials: "same-origin"`
- Endpoint: `$NEXT_PUBLIC_BASE_URL/api/graphql`

#### Server-Side Setup (apolloClient.ts):

```typescript
// Use for: Suspense data preloading, RSC integration
export const { getClient, query, PreloadQuery } = registerApolloClient(
  async () => {
    // Get cookies asynchronously
    const cookie = await cookies();
    const cookieString = cookie.toString();

    // Create context link to inject cookies
    const contextLink = new SetContextLink((prevContext, _operation) => {
      return {
        headers: {
          ...prevContext.headers,
          cookie: cookieString,
        },
      };
    });

    // Create HTTP link
    const httpLink = new HttpLink({
      uri: process.env.NEXT_PUBLIC_BASE_URL + "/api/graphql",
      credentials: "same-origin",
      fetchOptions: { cache: "no-store" },
    });

    // Concat links
    const link = contextLink.concat(httpLink);

    return new ApolloClient({
      cache: new InMemoryCache(),
      link,
    });
  },
);
```

**Available Exports**:

- `getClient()` - Get Apollo client instance
- `query(document, variables)` - Execute query with RSC
- `PreloadQuery` - Component for preloading in Suspense

#### Cache Strategy:

- **No persistence** - InMemoryCache only
- **No HTTP caching** - `no-store` on all requests
- **Always fresh** - Best for real-time dashboards
- **Apollo cache updates** - Manual cache updates on mutations (if configured)

## 7. DATA FLOW & PRELOADING PATTERNS

### A. Current Data Flow (Single-Page Pattern)

```
User Action
    ↓
Feature Hook (useEmployeeList)
    ├─ State: Gets page, limit, filter from Zustand
    └─ TODO: Call entity hook → const { data } = useGetEmployees(variables)
         ↓
    Entity Hook (useGetEmployee)
         ├─ Wraps Apollo useQuery
         └─ Document: GetEmployeeDocument (from @/shared/graphql/generated)
              ↓
    Apollo Client
         ├─ Check InMemoryCache
         └─ If miss → HTTP request → /api/graphql
              ↓
    Server Resolver
         ├─ Authentication check
         ├─ Prisma query
         └─ Return typed response
              ↓
    Widget Component (EmployeeTable)
    ├─ Receives: data, loading, error from hook
    ├─ Renders: Rows, pagination, sorting, filtering
    └─ Side effect: on edit/delete → Call mutation hook
```

### Codegen Workflow

```

3. Generates:
   - shared/graphql/generated/graphql.ts (types)
   - shared/graphql/generated/gql.ts (documents)
   - server/graphql/types/generated.ts (resolvers)

4. Entity hooks import generated types:
```

### Common Usage Patterns

```typescript
// Client Component with Query
"use client";
import { useGetEmployees } from "@/entities/core/employee";

export function EmployeesPage() {
  const { data, loading, error } = useGetEmployees({
    filter: { status: "active" },
    pagination: { skip: 0, take: 50 },
  });

  if (loading) return <Skeleton />;
  if (error) return <ErrorMessage />;

  return (
    <div>
      {data.employees.nodes.map(emp => (
        <EmployeeCard key={emp.id} {...emp} />
      ))}
    </div>
  );
}

// Client Component with Mutation
"use client";
import { useCreateEmployee } from "@/entities/core/employee";

export function CreateEmployeeForm() {
  const [createEmployee, { loading }] = useCreateEmployee();

  const handleSubmit = async (formData) => {
    await createEmployee({
      variables: { input: formData },
      refetchQueries: ["GetEmployees"], // Or manual cache update
    });
  };

  return <Form onSubmit={handleSubmit} disabled={loading} />;
}

// Client Component with Suspense
"use client";
import { Suspense } from "react";
import { useSuspenseQueryProcesses } from "@/entities/operations/process";

function ProcessesList({ companyId }: { companyId: string }) {
  const { data } = useSuspenseQueryProcesses(companyId);

  return (
    <ul>
      {data.processes.nodes.map(proc => (
        <li key={proc.id}>{proc.name}</li>
      ))}
    </ul>
  );
}

export function ProcessesPage({ companyId }: { companyId: string }) {

  );
}
```
